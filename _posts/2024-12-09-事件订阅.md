ActionWrapper类
-------------

订阅/注销事件就和 new/delete 对象一样, 必须是成对存在的.   

{% highlight csharp %}
event Action SunRise;

SomeEvents.SunRise += Getup;
SomeEvents.SunRise -= Getup;
{% endhighlight %}

使用 ActionWrapper 类将订阅/注销事件抽象出来  

{% highlight csharp %}
public class ActionWrapper{
    public Action<Action> Subscribe;
    public Action<Action> UnSubscribe;
}

ActionWrapper actionWrapper = new ActionWrapper();
actionWrapper.Subscribe = callback => SomeEvents.SunRise += callback;
actionWrapper.UnSubscribe = callback => SomeEvents.SunRise -= callback;

public class Link{
    public void OnEventRaised()
    {

    }

    public void Enable()
    {
        actionWrapper.Subscribe(OnEventRaised);
    }

    public void Disable()
    {
        actionWrapper.UnSubscribe(OnEventRaised);
    }
}
{% endhighlight %}
